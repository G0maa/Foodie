class UpdateCartItemController {
    constructor( useCase: UpdateCartItemUseCase) {}

    @AuthDecorators
    @ApiDecorators
    handle(command: UpdateCartItemCommand) {
        return this.useCase.execute(command);
    }
}


class UpdateCartItemUseCase {

    constructor(restaurantService: RestaurantService, cartService: CartService, repository: UpdateCartItemRepository) {}

    execute(command: UpdateCartItemCommand, userUuid) {

        // error if anything fails
        restaurantService.isRestaurantOpen(command.restaurantId)
        restaurantService.isItemAvailable(command.itemId)

        // error if a cart already exists => prompt to clear cart first
        cartService.verifyCartExists(command.cartId)
        repository.updateCartItem(command, userUuid)
        cartService.mergeItems()
        cartService.deleteZeroQuantityItems()
        cartService.recalculateTotals()

        return

    }
}
